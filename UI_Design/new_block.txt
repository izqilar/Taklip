/* ── 对象选择器：搜索/选择具体代理商或服务商 ── */
function syncObjUI(){
  const on=isViewAs();
  $('#objPick').style.display=on?'flex':'none';
  if(!on)return;
  const o=CUR.obj;
  $('#objQ').value='';
  $('#objTag').textContent=o.name+' · '+o.region+' · '+o.phone;
  $('#objNote').textContent='以「'+o.name+'」'+objTagName()+'身份查看 · 数据仅限该对象 · 总台'+(RO?'只读':'可写');
  renderObjList();
}
function renderObjList(){
  const q=$('#objQ').value.trim();
  const src=objSource().filter(o=>!q||[o.id,o.name,o.nick,o.phone,o.region].some(s=>s.includes(q)));
  $('#objList').innerHTML=src.length?src.map(o=>`<button class="o" data-id="${o.id}">
    <b>${o.name}</b><span>${o.nick?'昵称 '+o.nick+' · ':''}${o.region} · ${o.phone}</span>
    <span class="n">${o.id}${o===CUR.obj?' · 当前':''}</span></button>`).join('')
    :`<div class="o" style="cursor:default;color:var(--ink-3)">无匹配对象</div>`;
}
function setCurObj(o){CUR.obj=o;renderScope();syncObjUI();}
$('#objQ').addEventListener('focus',()=>{$('#objList').classList.add('on');renderObjList();});
